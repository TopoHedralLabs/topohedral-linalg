1//! Shared types and conversions for the BLAS/LAPACK wrapper layer.
2//!
3//! Provides the [`AsI32`] conversion trait used across the BLAS/LAPACK wrappers to convert Rust
4//! floating-point dimension and parameter values into the `i32` integers that LAPACK routines
5//! expect. Implementations are provided for `f32` and `f64`. Keeping this in a dedicated module
6//! avoids repetition and provides a single place to adjust the conversion strategy if the
7//! underlying BLAS/LAPACK ABI ever changes.

19//{{{ trait: AsI32
20/// Conversion trait for casting a floating-point value to `i32` for use in BLAS/LAPACK calls.
21pub trait AsI32 {
22    /// Returns the value cast to `i32`.
23    fn as_i32(&self) -> i32;
24}
25//{{{ impl: AsI32 for f32
26impl AsI32 for f32 {
27    fn as_i32(&self) -> i32 {
28        *self as i32
29    }
30}
31//}}}
32//{{{ impl: AsI32 for f64
33impl AsI32 for f64 {
34    fn as_i32(&self) -> i32 {
35        *self as i32
36    }
37}

42/// Converts a Rust matrix dimension to the integer type used by BLAS/LAPACK.
43///
44/// # Panics
45///
46/// Panics when `value` exceeds the range supported by the linked BLAS/LAPACK ABI.
47#[inline]
48pub(crate) fn blas_dim(
49    name: &str,
50    value: usize,
51) -> i32 {
52    i32::try_from(value)
53        .unwrap_or_else(|_| panic!("{name} exceeds the maximum BLAS/LAPACK dimension"))
54}
55
56/// Computes a matrix buffer length without allowing `usize` overflow.
57///
58/// # Panics
59///
60/// Panics when `nrows * ncols` cannot be represented by `usize`.
61#[inline]
62pub(crate) fn matrix_len(
63    name: &str,
64    nrows: usize,
65    ncols: usize,
66) -> usize {
67    nrows
68        .checked_mul(ncols)
69        .unwrap_or_else(|| panic!("{name} dimensions overflow usize"))
70}
71
72/// Verifies that a column-major buffer has exactly the expected matrix length.
73///
74/// # Panics
75///
76/// Panics when the dimensions overflow or `actual` differs from the expected length.
77#[inline]
78pub(crate) fn assert_matrix_len(
79    name: &str,
80    actual: usize,
81    nrows: usize,
82    ncols: usize,
83) {
84    let expected = matrix_len(name, nrows, ncols);
85    assert_eq!(
86        actual, expected,
87        "{name} buffer length does not match its dimensions"
88    );
89}

91/// Validates a workspace size returned by a LAPACK workspace query.
92///
93/// # Panics
94///
95/// Panics when LAPACK returns a non-positive workspace size.
96#[inline]
97pub(crate) fn workspace_len<T>(value: &T) -> (usize, i32)
98where
99    T: AsI32,
100{
101    let lwork = value.as_i32();
102    assert!(
103        lwork > 0,
104        "LAPACK workspace query returned a non-positive length"
105    );
106    (lwork as usize, lwork)
107}
