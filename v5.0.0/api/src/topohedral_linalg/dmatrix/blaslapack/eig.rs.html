1//! General (non-symmetric) eigendecomposition of a [`DMatrix`] via LAPACK `dgeev`/`sgeev`.
2//!
3//! Provides the `eig()` method on [`DMatrix<T>`], computing all eigenvalues and both the left
4//! and right eigenvectors of a general square matrix. The computation is delegated to the
5//! `Geev` LAPACK driver. Eigenvalues are returned as complex numbers even when the input is
6//! real-valued; eigenvector matrices are stored column-major in the `Return<T>` struct. LAPACK
7//! errors propagate as a typed `Error`.
8//--------------------------------------------------------------------------------------------------
9
10//{{{ crate imports
11use crate::blaslapack::{eig_raw, EigRawError, LapackScalar};
12use crate::common::Complex;
13use crate::dmatrix::DMatrix;
14//}}}
15//{{{ dep imports
16use thiserror::Error;
17//}}}
18//--------------------------------------------------------------------------------------------------
19
20//{{{ enum: Error
21/// Errors that can occur during general eigendecomposition.
22#[derive(Clone, Error, Debug, PartialEq, Eq)]
23#[non_exhaustive]
24pub enum Error {
25    /// The QR algorithm did not converge.
26    #[error("eigendecomposition did not converge")]
27    NoConvergence,
28    /// The validated backend call unexpectedly rejected an argument.
29    #[error("eigendecomposition backend failed with info code {info}")]
30    BackendFailure {
31        /// Raw LAPACK diagnostic code.
32        info: i32,
33    },
34}
35//}}}
36
37//{{{ struct: Return
38/// Represents the eigenvalue decomposition of a square matrix of size `N`.
39///
40/// The eigenvalue decomposition of a matrix `A` is a factorization of the form `A = PDP^-1`,
41/// where `P` is the matrix of right eigenvectors, `D` is the diagonal matrix of eigenvalues,
42/// and `P^-1` is the matrix of left eigenvectors.
43///
44/// This struct contains the left and right eigenvectors, as well as the eigenvalues, of the
45/// decomposition.
46#[derive(Clone, Debug)]
47pub struct Return<T> {
48    /// Matrix whose columns are the left eigenvectors of A.
49    pub left_eigvecs: DMatrix<T>,
50    /// Matrix whose columns are the right eigenvectors of A.
51    pub right_eigvecs: DMatrix<T>,
52    /// Eigenvalues as complex numbers (real part from `wr`, imaginary part from `wi`).
53    pub eigvals: Vec<Complex<T>>,
54}
55//}}}
56
57//{{{ impl DMatrix<T>
58impl<T> DMatrix<T>
59where
60    T: LapackScalar,
61{
62    /// Computes the eigendecomposition of a general square matrix.
63    ///
64    /// Returns all eigenvalues and both left and right eigenvectors of `self`. Eigenvalues are
65    /// represented as complex numbers even when the input matrix is real-valued.
66    ///
67    /// # Errors
68    ///
69    /// Returns [`Error::NoConvergence`] if the QR algorithm does not converge, or
70    /// [`Error::BackendFailure`] for an unexpected backend failure.
71    /// # Panics
72    ///
73    /// Panics if the matrix is not square or its dimensions exceed the LAPACK integer range.
74    pub fn eig(self) -> Result<Return<T>, Error> {
75        let n = self.nrows;
76        assert_eq!(
77            n, self.ncols,
78            "matrix must be square for eigendecomposition"
79        );
80        let raw = eig_raw(self.data, n).map_err(|error| match error {
81            EigRawError::LapackError(info) if info > 0 => Error::NoConvergence,
82            EigRawError::LapackError(info) => Error::BackendFailure { info },
83        })?;
84        Ok(Return {
85            left_eigvecs: DMatrix {
86                data: raw.vl,
87                nrows: n,
88                ncols: n,
89            },
90            right_eigvecs: DMatrix {
91                data: raw.vr,
92                nrows: n,
93                ncols: n,
94            },
95            eigvals: raw.eigvals,
96        })
97    }
98}
99//}}}
